<pre id="log">等待操作...</pre>
<script>
const log = m => document.getElementById('log').textContent = m + '\n' + document.getElementById('log').textContent;

if (!('NDEFReader' in window)) log('此瀏覽器不支援 Web NFC');

document.getElementById('scan').onclick = async () => {
  try {
    const ndef = new NDEFReader();
    await ndef.scan();
    log('請將標籤靠近手機背面...');
    ndef.onreadingerror = () => log('讀取失敗，可能不是 NDEF 標籤');
    ndef.onreading = ({ message, serialNumber }) => {
      log('序號：' + serialNumber);
      for (const r of message.records) {
        const data = r.data ? new TextDecoder(r.encoding || 'utf-8').decode(r.data) : '(空)';
        log(`類型：${r.recordType}｜內容：${data}`);
      }
    };
  } catch (e) { log('錯誤：' + e.name + ' ' + e.message); }
};

document.getElementById('write').onclick = async () => {
  const v = document.getElementById('txt').value.trim();
  if (!v) return log('請先輸入內容');
  try {
    const ndef = new NDEFReader();
    log('請將標籤靠近手機以寫入...');
    const isUrl = /^https?:\/\//.test(v);
    await ndef.write({ records: [{ recordType: isUrl ? 'url' : 'text', data: v }] });
    log('寫入成功：' + v);
  } catch (e) { log('錯誤：' + e.name + ' ' + e.message); }
};
</script>
